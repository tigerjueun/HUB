pje0914@ParkJuEun:~ $ echo "===== CREATE AEGIS ETHERNET PROFILE ====="

if nmcli -t -f NAME connection show | grep -Fxq "aegis-eth0"; then
    echo "aegis-eth0 already exists - updating"
    sudo nmcli connection modify aegis-eth0 \
        connection.interface-name eth0 \
        connection.autoconnect yes \
        connection.autoconnect-priority 100 \
        ipv4.method manual \
        ipv4.addresses 192.168.137.2/24 \
        ipv4.never-default yes \
        ipv6.method disabled
else
    echo "Creating aegis-eth0"
    sudo nmcli connection add \
        type ethernet \
        ifname eth0 \
        con-name aegis-eth0 \
        connection.autoconnect yes \
        connection.autoconnect-priority 100 \
        ipv4.method manual \
        ipv4.addresses 192.168.137.2/24 \
        ipv4.never-default yes \
        ipv6.method disabled
fi

echo
echo "===== ACTIVATE ====="

sudo nmcli connection down netplan-eth0 2>/dev/null || true
sudo nmcli connection up aegis-eth0
