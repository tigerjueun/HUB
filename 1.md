echo
echo "========== 1. OS =========="
cat /etc/os-release | grep -E 'PRETTY_NAME|VERSION='
uname -m
python3 --version

echo
echo "========== 2. HOST =========="
hostname
hostnamectl

echo
echo "========== 3. TIME =========="
timedatectl | head -n 10

echo
echo "========== 4. CAMERA CONFIG =========="
sudo grep -nE \
'camera_auto_detect|dtoverlay=(imx219|imx708|ov5647|imx477|imx296)' \
/boot/firmware/config.txt 2>/dev/null || true

echo
echo "========== 5. CAMERA TOOLS =========="
command -v rpicam-hello || echo "rpicam-hello NOT FOUND"
command -v rpicam-vid || echo "rpicam-vid NOT FOUND"

echo
echo "========== 6. DETECTED CAMERAS =========="
rpicam-hello --list-cameras 2>&1

echo
echo "========== 7. NETWORK =========="
ip -br addr

echo
echo "========== 8. ROUTE =========="
ip route

echo
echo "========== 9. NETWORKMANAGER =========="
nmcli -f DEVICE,TYPE,STATE,CONNECTION device status 2>/dev/null || echo "nmcli NOT FOUND"

echo
echo "========== 10. SSH =========="
systemctl is-enabled ssh 2>/dev/null || true
systemctl is-active ssh 2>/dev/null || true
