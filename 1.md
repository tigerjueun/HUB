socket = context.socket(zmq.PUB)

# AEGIS realtime video:
# keep a small bounded PUB queue and let stale frames drop naturally.
# Do not use ZMQ_CONFLATE here; it overrides HWM semantics.
socket.setsockopt(zmq.LINGER, 0)
socket.setsockopt(zmq.SNDHWM, 2)

try:
    socket.setsockopt(zmq.SNDBUF, 1024 * 1024)
except zmq.ZMQError:
    pass
